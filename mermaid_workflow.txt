sequenceDiagram
    autonumber
    
    %% Actors and Components
    actor Emp as Employee
    box Frontend (Vue 3 / Vercel)
        participant UI as GoalSheet.vue
        participant Pinia as stores/goals.js
        participant API as services/api.js
    end
    box Backend (Flask / Railway)
        participant Flask as routes/employee.py
        participant DB as models/goal.py (SQLite)
    end
    actor Mgr as Manager

    %% -------------------------------------
    %% FLOW 1: Drafting Goals
    %% -------------------------------------
    rect rgb(238, 242, 255)
    note right of Emp: Flow 1: Goal Drafting & Validation
    Emp->>UI: Types Target & Weightage
    UI->>Pinia: Updates local state
    Pinia-->>UI: Calculates Total Weightage
    opt If Weightage != 100%
        UI-->>Emp: Disables "Submit" button
    end
    Emp->>UI: Clicks "Submit Goal Sheet"
    UI->>API: POST /api/goals/sheet/{id}/submit
    API->>Flask: Validates request & JWT
    Flask->>DB: Updates sheet status to 'submitted'
    DB-->>Flask: Confirms mutation
    Flask-->>UI: 200 OK (New Sheet State)
    UI-->>Emp: Shows "Awaiting Approval" UI
    end

    %% -------------------------------------
    %% FLOW 2: Manager Review
    %% -------------------------------------
    rect rgb(240, 253, 244)
    note right of Mgr: Flow 2: Manager Approval
    Mgr->>UI: Navigates to TeamDashboard.vue
    UI->>API: GET /api/manager/team
    API->>Flask: routes/manager.py
    Flask->>DB: Fetch users where manager_id = {Mgr.id}
    DB-->>Flask: Return Team Data
    Flask-->>UI: 200 OK (Team List)
    Mgr->>UI: Reviews & Clicks "Approve"
    UI->>API: POST /api/manager/sheet/{id}/approve
    API->>Flask: Updates status to 'approved'
    Flask->>DB: Locks sheet (cannot be edited)
    DB-->>Flask: 
    Flask-->>UI: 200 OK
    end

    %% -------------------------------------
    %% FLOW 3: Unlock Request
    %% -------------------------------------
    rect rgb(254, 242, 242)
    note right of Emp: Flow 3: The Unlock Workflow
    Emp->>UI: Clicks "Request Changes"
    UI->>API: POST /api/goals/sheet/{id}/request-unlock
    API->>Flask: routes/employee.py
    Flask->>DB: Sets unlock_requested = True
    DB-->>Flask:
    Flask-->>UI: 200 OK
    
    Mgr->>UI: Types Feedback & clicks "Accept & Unlock"
    UI->>API: POST /api/manager/sheet/{id}/resolve-unlock
    API->>Flask: routes/manager.py
    Flask->>DB: Sets status = 'draft'
    DB-->>Flask:
    Flask-->>UI: 200 OK
    UI-->>Emp: Sheet is editable again!
    end
